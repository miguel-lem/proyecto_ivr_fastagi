#2.-la segunda version esta pulida y corregida el error de no tener el control de tiempo que ha timbrado
#con eso se puede conocer en caso que las llamadas hayan sido colgadas porque no quieren contestar
exten => 401,1,NoOp(Llamada a la extensión 401)
 same => n,Set(NUM_MARCADO=401)
 same => n,Set(TIME_INICIO=${EPOCH});<---el epoch permite controlar a nivel de sistema el momento que empieza
 same => n,Dial(PJSIP/401,25,g)
 same => n,Set(TIME_FIN=${EPOCH});<---aqui controlo el siguiente tiempo y puedo con eso operar para saber el tiempo de timbrado
 same => n,Set(DIALEDTIME=$[${TIME_FIN} - ${TIME_INICIO}])
 same => n,Hangup()

exten => h,1,NoOp(--- Llegada al hangup handler ---)
 same => n,NoOp(DIALSTATUS=${DIALSTATUS})
 same => n,NoOp(DIALEDTIME=${DIALEDTIME})
 same => n,Set(DIALEDTIME=${IF($[${ISNULL(${DIALEDTIME})}]?0:${DIALEDTIME})})
 same => n,Set(ANSWEREDTIME=${IF($[${ISNULL(${ANSWEREDTIME})}]?0:${ANSWEREDTIME})})
 same => n,Set(TIMBRADO=$[${DIALEDTIME} - ${ANSWEREDTIME}])
 same => n,Set(LLAMADA_DATA=${NUM_MARCADO}|${DIALSTATUS}|${TIMBRADO}|${ANSWEREDTIME}|${UNIQUEID})
 same => n,NoOp(LLAMADA_DATA=${LLAMADA_DATA})
 same => n,AGI(agi://192.168.2.10:4574/conAgiestados?${LLAMADA_DATA})
